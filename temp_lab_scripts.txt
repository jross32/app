<script>
  // Helper to render tables
  function renderSimTable(rows) {
    if (!rows || !rows.length) return '<div class="text-secondary small">No data.</div>';
    return `
      <table class="table table-dark align-middle mb-0">
        <thead>
          <tr><th>Team</th><th>Current</th><th>Top 3</th><th>1st</th></tr>
        </thead>
        <tbody>
          ${rows.map(r => `
            <tr class="${r.cue ? 'table-success' : ''}">
              <td>${r.team}</td>
              <td>${r.current}</td>
              <td>${r.pct_top3}%</td>
              <td>${r.pct_first}%</td>
            </tr>
          `).join('')}
        </tbody>
      </table>`;
  }

  const predictorForm = document.getElementById('predictorForm');
  const predictorResult = document.getElementById('predictorResult');
  if (predictorForm && predictorResult) {
    predictorForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      const formData = new FormData(predictorForm);
      const btn = predictorForm.querySelector('button[type="submit"]');
      if (btn) { btn.disabled = true; btn.innerText = 'Predicting...'; }
      try {
        const res = await fetch('/lab/predict', { method: 'POST', body: formData });
        const data = await res.json();
        if (!res.ok || data.error) throw new Error(data.error || 'Error');
        predictorResult.innerHTML = `
          <div class="d-flex justify-content-between">
            <div><strong>${data.our_team}</strong>: ${data.our_rating}</div>
            <div><strong>${data.opp_team}</strong>: ${data.their_rating}</div>
          </div>
          <div class="subtle mt-1">Projected: ${data.our_team} ${data.proj_us} – ${data.opp_team} ${data.proj_them}</div>
          <div class="text-info mt-1">${data.summary}</div>
          ${(data.reasons && data.reasons.length) ? `<ul class="small text-secondary mt-1 mb-0">${data.reasons.map(r=>`<li>${r}</li>`).join('')}</ul>` : ''}
          <div class="small text-secondary mt-2">
            Race-based advantage: ${(data.ours?.components?.race_eff || 0).toFixed(2)} vs ${(data.theirs?.components?.race_eff || 0).toFixed(2)}<br>
            Hill-hill confidence: ${(data.ours?.components?.hill_conf || 0).toFixed(2)} vs ${(data.theirs?.components?.hill_conf || 0).toFixed(2)}<br>
            SL matchup difficulty factored by opponent roster.
          </div>
        `;
      } catch (err) {
        predictorResult.innerHTML = `<div class="text-danger small">Prediction failed.</div>`;
      } finally {
        if (btn) { btn.disabled = false; btn.innerText = 'Predict'; }
      }
    });
  }

  const simulateForm = document.getElementById('simulateForm');
  const simResult = document.getElementById('simResult');
  if (simulateForm && simResult) {
    simulateForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      const formData = new FormData(simulateForm);
      const btn = simulateForm.querySelector('button[type="submit"]');
      if (btn) { btn.disabled = true; btn.innerText = 'Running...'; }
      try {
        const res = await fetch('/lab/simulate', { method: 'POST', body: formData });
        const data = await res.json();
        if (!res.ok || data.error) throw new Error(data.error || 'Error');
        simResult.innerHTML = renderSimTable(data.rows);
      } catch (err) {
        simResult.innerHTML = `<div class="text-danger small">Simulation failed.</div>`;
      } finally {
        if (btn) { btn.disabled = false; btn.innerText = 'Run Simulation'; }
      }
    });
  }

  const lineupForm = document.getElementById('lineupForm');
  const lineupResult = document.getElementById('lineupResult');
  const lineupTeam = document.getElementById('lineupTeam');
  const lineupFormat = document.getElementById('lineupFormat');
  const lineupRosterTable = document.getElementById('lineupRosterTable');
  if (lineupForm && lineupResult) {
    async function fetchRoster() {
      if (!lineupTeam || !lineupFormat || !lineupRosterTable) return;
      const fd = new FormData();
      fd.append('team_id', lineupTeam.value);
      fd.append('format', lineupFormat.value);
      try {
        const res = await fetch('/lab/roster', { method: 'POST', body: fd });
        const data = await res.json();
        if (!res.ok || data.error) throw new Error();
        const roster = data.roster || [];
        const rows = roster.map(p => `
          <tr>
            <td><input class="form-check-input" type="checkbox" name="include_player" value="${p.id}" form="lineupForm" checked></td>
            <td><input class="form-check-input" type="checkbox" name="lock_player" value="${p.id}" form="lineupForm"></td>
            <td>${p.name}</td>
            <td>${p.sl}</td>
            <td>${(p.ppm||0).toFixed(2)}</td>
            <td>${((p.win_pct||0)*100).toFixed(1)}%</td>
            <td>${((p.pa||0)*100).toFixed(1)}%</td>
            <td>${p.role||'Contributor'}</td>
          </tr>
        `).join('');
        lineupRosterTable.innerHTML = `
          <table class="table table-dark table-sm align-middle mb-0">
            <thead><tr><th>Include</th><th>Lock</th><th>Player</th><th>SL</th><th>PPM</th><th>Win%</th><th>PA%</th><th>Role</th></tr></thead>
            <tbody>${rows}</tbody>
          </table>`;
      } catch (err) {
        lineupRosterTable.innerHTML = `<div class="text-danger small">Unable to load roster.</div>`;
      }
    }

    if (lineupFormat) lineupFormat.addEventListener('change', fetchRoster);

          lineupResult.innerHTML = `<div class="text-secondary small">No valid lineup for this cap.</div>`;

                        <td>${p.sl}</td>
                        <td>${p.ppm.toFixed(2)}</td>
                        <td>${(p.pa*100).toFixed(1)}%</td>

                </table>
              </div>
              <div class="text-secondary small mt-1">Highest lineup score while meeting SL cap and locks.</div>
            </div>`;

          html += `<div class="text-secondary small mb-2">Alternatives</div><div class="scroll-area">`;
          alts.forEach((line, idx) => {
            html += `
              <div class="mb-2 p-2 rounded" style="background:rgba(255,255,255,.04);">
                <div class="d-flex justify-content-between">
                  <strong>#${idx+2} Score: ${line.score.toFixed(2)}</strong>
                  <span class="subtle">SL: ${line.sl_total} | PPM: ${line.ppm_avg.toFixed(2)} | PA: ${(line.pa_avg*100).toFixed(1)}%</span>
                </div>
                <div class="table-responsive mt-1">
                  <table class="table table-dark table-sm align-middle mb-0">
                    <thead><tr><th>Player</th><th>SL</th></tr></thead>
                    <tbody>
                      ${line.players.map(p=>`<tr><td>${p.name}</td><td>${p.sl}</td></tr>`).join('')}
                    </tbody>
                  </table>
                </div>
              </div>`;
          });
          html += `</div>`;

      } catch (err) {
        lineupResult.innerHTML = `<div class="text-danger small">Lineup failed.</div>`;
      } finally {
        if (btn) { btn.disabled = false; btn.innerText = 'Generate'; }
      }
    });

    fetchRoster();
  }

              </div>
            </div>

                <div class="subtle">${data.status}</div>
                <div class="text-secondary small">Recent: Win% ${(data.recent.win_pct*100).toFixed(1)}% | PPM ${data.recent.ppm.toFixed(2)}</div>

            </div>
          </div>
          <div class="row g-2 mt-2">
            <div class="col-12 col-md-4"><div class="p-2 rounded" style="background:rgba(255,255,255,.04);"><div class="text-secondary small">Vs Lower SL</div><div class="subtle">Win% ${((band.lower?.win_pct||0)*100).toFixed(0)}% | PPM ${(band.lower?.ppm||0).toFixed(1)}</div></div></div>
